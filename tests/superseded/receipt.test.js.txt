'use strict';
const test = require('node:test');
const assert = require('node:assert/strict');
const fs = require('node:fs');
const path = require('node:path');
const { JSDOM } = require('jsdom');
const { receiptReferences } = require('../Bouncebash_CS/receipt-scan');

const source = name => fs.readFileSync(path.join(__dirname, '..', 'Bouncebash_CS', name), 'utf8');
const nextTick = () => new Promise(resolve => setImmediate(resolve));

test('receipt reference candidates depend on receipt labels and value shape', () => {
  assert.deepEqual(receiptReferences('GCash\nReference No. 0048 3948 0938\nAmount: PHP 450.00'), ['0048 3948 0938']);
  assert.deepEqual(receiptReferences('Transfer ID:\nBPI-6a7zP092\nMerchant ID: 47892381'), ['BPI-6a7zP092']);
  assert.deepEqual(receiptReferences('Transaction ID: TX-5571\nReference # TX-5571\nReceipt no: CX/73141'), ['TX-5571', 'CX/73141']);
  assert.deepEqual(receiptReferences('Account Number: 47102458\nReference: 2026-11-21\nReference: 09123456789\nReference: 550.00\nMerchant ID: 77473'), []);
});

test('OCR suggests, asks, falls back, and respects edits or a stale image', async t => {
  const { window } = new JSDOM('<input id="ref"><p id="status"></p><div id="choices" hidden><select></select></div>', { runScripts: 'outside-only' });
  t.after(() => window.close());
  const reference = window.document.querySelector('#ref');
  const status = window.document.querySelector('#status');
  const choices = window.document.querySelector('#choices');
  const results = [];
  let workers = 0;
  let terminated = 0;
  window.Tesseract = { createWorker: async () => {
    workers++;
    return { recognize: () => new Promise(resolve => results.push(resolve)), terminate: async () => { terminated++; } };
  } };
  window.eval(source('receipt-scan.js') + '\nwindow.startScan = initReceiptScan;');
  const scanner = window.startScan(reference, status, choices);
  async function start() { scanner.scan({}); await nextTick(); }
  async function finish(text, confidence = 99) {
    results.shift()({ data: { text, confidence } });
    await nextTick();
  }
  await start();
  assert.match(status.textContent, /Scanning/);
  await finish('Transaction ID: TX-12345');
  assert.equal(reference.value, 'TX-12345');
  assert.match(status.textContent, /Check it/);
  scanner.reset();

  await start();
  await finish('Reference: FIRST-1\nTransaction ID: SECOND-2');
  assert.equal(choices.hidden, false);
  assert.equal(reference.value, '');
  const select = choices.querySelector('select');
  select.value = 'SECOND-2';
  select.dispatchEvent(new window.Event('change'));
  assert.equal(reference.value, 'SECOND-2');
  scanner.reset();

  await start();
  await finish('Amount: 420.00');
  assert.match(status.textContent, /Enter it manually/);
  await start();
  await finish('Reference: TX-12345', 45);
  assert.equal(reference.value, '');

  await start();
  reference.value = 'MANUAL-42';
  reference.dispatchEvent(new window.Event('input'));
  await finish('Reference: TX-87654');
  assert.equal(reference.value, 'MANUAL-42');

  reference.value = '';
  await start();
  const count = workers;
  scanner.reset();
  await start();
  await finish('Reference: STALE-01');
  assert.equal(reference.value, '');
  assert.equal(workers, count);
  await finish('Reference: FRESH-02');
  assert.equal(reference.value, 'FRESH-02');
  assert.equal(workers, terminated);

  scanner.reset();
  delete window.Tesseract;
  await start();
  assert.match(status.textContent, /unavailable/);
});

test('the receipt print control is specific to verified payments and invokes window.print', t => {
  const { window } = new JSDOM('<div id="receipt"></div>', { runScripts: 'outside-only' });
  t.after(() => window.close());
  window.eval(source('customer-tools.js') + '\nwindow.showReceipt = renderReceipt;');
  const container = window.document.querySelector('#receipt');
  const b = { payments: [{ status: 'pending', amount_cents: 100, reference: 'TEST' }] };
  window.showReceipt(b, container);
  assert.equal(container.hidden, true);
  assert.equal(container.querySelector('button'), null);

  b.payments[0].status = 'verified';
  window.showReceipt(b, container);
  const button = container.querySelector('button.btn-receipt-print.no-print');
  assert.ok(button);
  assert.equal(button.textContent, 'Print receipt');
  assert.match(container.textContent, /not an official receipt/);
  let calls = 0;
  window.print = () => { calls++; };
  button.click();
  assert.equal(calls, 1);
});
